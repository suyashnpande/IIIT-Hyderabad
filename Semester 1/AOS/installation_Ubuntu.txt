# =========================================
# OpenCV Installation on Ubuntu/Debian
# =========================================

# ---- 1. Update package list ----
sudo apt update

# ---- 2. Install required build tools ----
sudo apt install -y build-essential cmake git pkg-config

# ---- 3. Install GUI and media I/O libs ----
sudo apt install -y \
    libgtk-3-dev \
    libavcodec-dev libavformat-dev libswscale-dev \
    libv4l-dev v4l-utils \
    libxvidcore-dev libx264-dev

# ---- 4. Install image I/O libraries ----
sudo apt install -y \
    libjpeg-dev libpng-dev libtiff-dev \
    libwebp-dev libopenexr-dev \
    libdc1394-dev

# ---- 5. Install Python (optional, for bindings) ----
sudo apt install -y python3-dev python3-pip python3-numpy

# ---- 6. Clone OpenCV ----
git clone https://github.com/opencv/opencv.git
git clone https://github.com/opencv/opencv_contrib.git

# ---- 7. Create build directory ----
cd opencv
mkdir build
cd build

# ---- 8. Run CMake ----
cmake -D CMAKE_BUILD_TYPE=Release \
      -D CMAKE_INSTALL_PREFIX=/usr/local \
      -D OPENCV_EXTRA_MODULES_PATH=../../opencv_contrib/modules \
      ..

# ---- 9. Compile ----
make -j$(nproc)

# ---- 10. Install system-wide ----
sudo make install

# ---- 11. Verify Installation ----
pkg-config --modversion opencv4

# ---- 12. Compile your C++ code ----
g++ your_code.cpp -o your_program `pkg-config --cflags --libs opencv4`
